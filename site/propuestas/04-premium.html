Donde ya trabajas.

Un script, un import, un componente de React o un elemento HTML para Angular, Vue y Svelte.

<div id="ventas"></div>
<script src="lozcharts.umd.js"></script>
<script>
  new LozChart('#ventas', {
    data: ventas,
    xAxis: { field: 'mes', type: 'time' },
    series: [{ type: 'line', field: 'online', name: 'Online' }],
  });
</script>
import { Chart } from 'lozcharts/react';

<Chart
  data={ventas}
  xAxis={{ field: 'mes', type: 'time' }}
  series={[{ type: 'bar', field: 'ventas', name: 'Ventas' }]}
  zoom
/>
import 'lozcharts/element';

<loz-chart [options]="opciones" (lozzoom)="alHacerZoom($event)"></loz-chart>
<div data-lozchart="#opciones-ventas"></div>
{{ opciones|json_script:"opciones-ventas" }}
<script>
  htmx.onLoad((el) => LozChart.initAll(el));
</script>
